What is HPC?
  HPC = High Performance Computing
  Making programs run FASTER by using multiple CPU cores
  at the same time.

  Normal:    1 core does all the work
  Parallel:  4 cores share the work = ~4x faster

What is OpenMP?
  OpenMP = Open Multi-Processing
  A library for parallel programming in C/C++.
  You add #pragma omp comments and the compiler
  automatically splits work across threads.

  Compile command: g++ -fopenmp filename.cpp -o output

What is a Thread?
  A thread is a lightweight worker.
  4 cores = 4 threads can run simultaneously.
  Each thread does a portion of the work.

What is a Race Condition?
  When 2 threads read/write the same variable at same time.
  Example:
    sum = 0
    Thread 1 reads sum=0, adds 5
    Thread 2 reads sum=0, adds 3
    Thread 1 writes sum=5
    Thread 2 writes sum=3
    Final sum = 3 (WRONG! Should be 8)
  Fix: use critical section or reduction.


KEY OPENMP COMMANDS (MEMORIZE THESE):

  #pragma omp parallel for
    --> splits a for-loop across multiple threads

  #pragma omp critical
    --> only ONE thread can enter this block at a time
    --> like a lock on a door

  #pragma omp parallel sections
    --> runs different code blocks on different threads

  #pragma omp section
    --> marks one block inside parallel sections

  #pragma omp task
    --> creates independent job for any free thread

  #pragma omp single
    --> only ONE thread runs this, others skip

  #pragma omp parallel for reduction(+:sum)
    --> each thread gets private copy, combines at end



===============================================================
  PRACTICAL 1: PARALLEL BFS AND DFS
===============================================================

AIM: Traverse a graph using BFS and DFS in parallel with OpenMP.

WHAT IS A GRAPH?
  Vertices (nodes) connected by Edges (lines).
  Example:
         0
        / \
       1   2
      / \   \
     3   4   5

  Vertices: 0, 1, 2, 3, 4, 5
  Edges: 0-1, 0-2, 1-3, 1-4, 2-5

BFS (Breadth-First Search):
  Visit level by level (all neighbors first)
  Uses: Queue (First In, First Out)
  Order: 0 -> 1, 2 -> 3, 4, 5

DFS (Depth-First Search):
  Go as deep as possible, then backtrack
  Uses: Recursion / Stack (Last In, First Out)
  Order: 0 -> 1 -> 3 -> 4 -> 2 -> 5


----------- CODE LINE BY LINE -----------


--- HEADERS ---

#include <iostream>
  --> for cout (printing) and cin (input)

#include <vector>
  --> for vector (resizable array)

#include <queue>
  --> for queue (FIFO, used in BFS)

#include <omp.h>
  --> OpenMP header, enables all #pragma omp

using namespace std;
  --> so we write cout instead of std::cout


--- GRAPH CLASS ---

class Graph {
    int V;
    vector<vector<int>> adj;
  --> V = number of vertices
  --> adj = adjacency list
  --> adj[0] = {1, 2} means vertex 0 connects to 1 and 2

    Graph(int V) {
        this->V = V;
        adj.resize(V);
    }
  --> constructor: sets V and creates V empty neighbor lists

    void addEdge(int u, int v) {
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
  --> adds edge between u and v
  --> adds BOTH directions because undirected graph


--- PARALLEL BFS ---

void parallelBFS(int start) {
    vector<bool> visited(V, false);
      --> track which nodes we visited (all start as false)

    queue<int> q;
      --> queue for BFS

    visited[start] = true;
    q.push(start);
      --> mark start node as visited, add to queue

    while (!q.empty()) {
        int size = q.size();
          --> how many nodes at current level

        #pragma omp parallel for
        for (int i = 0; i < size; i++) {
          --> process ALL nodes at this level IN PARALLEL
          --> if level has 3 nodes, 3 threads work at once

            int node = -1;
            #pragma omp critical
            {
                if (!q.empty()) {
                    node = q.front();
                    q.pop();
                    cout << node << " ";
                }
            }
          --> critical = only ONE thread pops from queue at a time
          --> WHY? queue is shared, 2 threads popping = data corruption
          --> gets front node, removes it, prints it

            if (node != -1) {
                for (int neighbor : adj[node]) {
                    #pragma omp critical
                    {
                        if (!visited[neighbor]) {
                            visited[neighbor] = true;
                            q.push(neighbor);
                        }
                    }
                }
            }
          --> for each neighbor of current node:
          --> check if visited (inside critical to prevent race condition)
          --> if not visited, mark it and add to queue
        }
    }
}


--- PARALLEL DFS ---

void parallelDFSUtil(int node, vector<bool>& visited) {
    bool alreadyVisited;
    #pragma omp critical
    {
        alreadyVisited = visited[node];
        if (!visited[node]) {
            visited[node] = true;
            cout << node << " ";
        }
    }
  --> safely checks and marks node as visited
  --> critical because visited[] is shared by all threads

    if (alreadyVisited) return;
  --> if already visited, stop - dont go deeper

    #pragma omp parallel for
    for (int i = 0; i < adj[node].size(); i++) {
        int neighbor = adj[node][i];
        if (!visited[neighbor]) {
            #pragma omp task
            parallelDFSUtil(neighbor, visited);
        }
    }
  --> for each unvisited neighbor, create a TASK
  --> #pragma omp task = independent job for any free thread
  --> all branches explored in parallel!
}

void parallelDFS(int start) {
    vector<bool> visited(V, false);
    #pragma omp parallel
    {
        #pragma omp single
        parallelDFSUtil(start, visited);
    }
}
  --> #pragma omp parallel = creates team of threads
  --> #pragma omp single = only ONE thread starts from root
  --> WHY single? Without it ALL threads start from root = waste
  --> tasks handle distributing work after that


--- MAIN ---

int main() {
    int V, E;
    cin >> V;       // 6
    Graph g(V);
    cin >> E;       // 5
    for (int i = 0; i < E; i++) {
        int u, v;
        cin >> u >> v;
        g.addEdge(u, v);
    }
    int start;
    cin >> start;   // 0
    g.parallelBFS(start);
    g.parallelDFS(start);
    return 0;
}

SAMPLE INPUT:
  6
  5
  0 1
  0 2
  1 3
  1 4
  2 5
  0

SAMPLE OUTPUT:
  Parallel BFS Traversal: 0 1 2 3 4 5
  Parallel DFS Traversal: 0 1 3 4 2 5

COMPILE: g++ -fopenmp whole.cpp -o bfs_dfs



===============================================================
  PRACTICAL 2: PARALLEL BUBBLE SORT AND MERGE SORT
===============================================================

AIM: Sort array using Bubble Sort and Merge Sort, both
     sequential and parallel, and compare their speeds.

WHAT IS BUBBLE SORT?
  Compare adjacent elements, swap if wrong order.
  Largest element "bubbles up" to the end each pass.
  Time: O(n^2) - slow for large arrays

WHAT IS MERGE SORT?
  Divide array in half, sort each half, merge them.
  Time: O(n log n) - much faster than bubble sort


----------- CODE LINE BY LINE -----------


--- SETUP ---

#include <cstdlib>    // for rand() - random numbers
#include <ctime>      // for time() - seed random
#include <omp.h>
#define SIZE 10000    // array size: 10,000 elements


--- SEQUENTIAL BUBBLE SORT ---

void bubbleSortSeq(vector<int>& arr) {
    int n = arr.size();
    for (int i = 0; i < n - 1; i++) {
        for (int j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                swap(arr[j], arr[j + 1]);
            }
        }
    }
}
  --> outer loop: n-1 passes
  --> inner loop: compare adjacent pairs, swap if left > right
  --> n-i-1: last i elements already sorted, skip them
  --> standard O(n^2) bubble sort


--- PARALLEL BUBBLE SORT (ODD-EVEN) ---

THE PROBLEM:
  In normal bubble sort, comparing (j,j+1) and (j+1,j+2)
  OVERLAP at index j+1. Cant do in parallel!

THE SOLUTION - ODD-EVEN TRANSPOSITION:
  EVEN phase: compare (0,1), (2,3), (4,5)  --> NO overlap!
  ODD phase:  compare (1,2), (3,4), (5,6)  --> NO overlap!

void bubbleSortParallel(vector<int>& arr) {
    int n = arr.size();
    for (int i = 0; i < n; i++) {

        // Even phase
        #pragma omp parallel for
        for (int j = 0; j < n - 1; j += 2) {
            if (arr[j] > arr[j + 1])
                swap(arr[j], arr[j + 1]);
        }
          --> j += 2: jump by 2, picks 0, 2, 4, 6...
          --> compares (0,1), (2,3), (4,5)
          --> no overlap so parallel is safe!

        // Odd phase
        #pragma omp parallel for
        for (int j = 1; j < n - 1; j += 2) {
            if (arr[j] > arr[j + 1])
                swap(arr[j], arr[j + 1]);
        }
          --> starts at j=1, picks 1, 3, 5, 7...
          --> compares (1,2), (3,4), (5,6)
          --> no overlap!
    }
}

  NO critical section needed because each thread works
  on completely different indices!


--- MERGE FUNCTION ---

void merge(vector<int>& arr, int l, int m, int r) {
    vector<int> left(arr.begin()+l, arr.begin()+m+1);
    vector<int> right(arr.begin()+m+1, arr.begin()+r+1);
      --> creates two temporary arrays: left half and right half

    int i = 0, j = 0, k = l;
    while (i < left.size() && j < right.size()) {
        if (left[i] <= right[j]) arr[k++] = left[i++];
        else arr[k++] = right[j++];
    }
      --> compare elements from both halves
      --> put smaller one into main array

    while (i < left.size()) arr[k++] = left[i++];
    while (j < right.size()) arr[k++] = right[j++];
      --> copy any remaining elements
}


--- SEQUENTIAL MERGE SORT ---

void mergeSortSeq(vector<int>& arr, int l, int r) {
    if (l < r) {
        int m = (l + r) / 2;
        mergeSortSeq(arr, l, m);       // sort left half
        mergeSortSeq(arr, m + 1, r);   // sort right half
        merge(arr, l, m, r);           // merge sorted halves
    }
}


--- PARALLEL MERGE SORT ---

void mergeSortParallel(vector<int>& arr, int l, int r, int depth) {
    if (l < r) {
        int m = (l + r) / 2;

        if (depth <= 0) {
            mergeSortSeq(arr, l, m);
            mergeSortSeq(arr, m + 1, r);
        }
          --> DEPTH LIMIT: when depth=0, stop creating threads
          --> go sequential to avoid thread overhead
          --> WHY? too many tiny threads = slower than sequential

        else {
            #pragma omp parallel sections
            {
                #pragma omp section
                mergeSortParallel(arr, l, m, depth - 1);

                #pragma omp section
                mergeSortParallel(arr, m + 1, r, depth - 1);
            }
        }
          --> parallel sections: run 2 blocks on DIFFERENT threads
          --> left half on Thread A, right half on Thread B
          --> depth-1: reduce depth each level

        merge(arr, l, m, r);
          --> after both halves sorted, merge them
    }
}

  HOW IT WORKS:
                [full array]         depth 4
               /            \
        [left half]    [right half]  depth 3 (2 threads)
        /      \        /      \
      [1/4]  [1/4]   [1/4]  [1/4]  depth 2 (4 threads)
      
  depth=4 means at most 2^4 = 16 parallel tasks


--- MAIN (TIMING) ---

int main() {
    vector<int> arr(SIZE), temp;
    srand(time(0));
    generateRandom(arr);
      --> creates 10,000 random numbers

    double start, end;

    temp = arr;                      // copy same unsorted data
    start = omp_get_wtime();         // start timer
    bubbleSortSeq(temp);
    end = omp_get_wtime();           // stop timer
    cout << "Sequential Bubble Sort Time: " << (end-start) << " sec";
      --> omp_get_wtime() = precise wall-clock timer
      --> temp = arr so each sort gets SAME unsorted data

    // same for parallel bubble, sequential merge, parallel merge...

    mergeSortParallel(temp, 0, SIZE - 1, 4);
      --> depth=4 allows up to 16 threads
}

TYPICAL OUTPUT:
  Sequential Bubble Sort Time: 0.45 sec
  Parallel Bubble Sort Time:   0.12 sec   <-- 3-4x faster!
  Sequential Merge Sort Time:  0.003 sec
  Parallel Merge Sort Time:    0.001 sec   <-- 2-3x faster!

COMPILE: g++ -fopenmp whole.cpp -o sort



===============================================================
  PRACTICAL 3: PARALLEL REDUCTION (MIN, MAX, SUM, AVG)
===============================================================

AIM: Find min, max, sum, and average of an array using
     OpenMP parallel reduction.

WHAT IS REDUCTION?
  Combining all elements into one result.
  
  Without OpenMP:
    sum = 0
    for each element: sum += element    (one by one, slow)

  With OpenMP reduction:
    Thread 1: adds elements 1-250    --> partial sum = 100
    Thread 2: adds elements 251-500  --> partial sum = 200
    Thread 3: adds elements 501-750  --> partial sum = 150
    Thread 4: adds elements 751-1000 --> partial sum = 180
    OpenMP combines: 100 + 200 + 150 + 180 = 630   (automatic!)


----------- CODE LINE BY LINE -----------

#include <iostream>
#include <omp.h>
using namespace std;


--- MIN VALUE ---

int minval(int arr[], int n) {
    int minval = arr[0];
      --> start by assuming first element is minimum

    #pragma omp parallel for reduction(min : minval)
    for (int i = 0; i < n; i++) {
        if (arr[i] < minval) minval = arr[i];
    }
      --> reduction(min : minval) does 3 things:
          1. Each thread gets its OWN private copy of minval
          2. Each thread finds minimum in its portion
          3. After loop, OpenMP takes MINIMUM of all results
      --> no race condition because each thread has own copy!

    return minval;
}


--- MAX VALUE ---

int maxval(int arr[], int n) {
    int maxval = arr[0];
    #pragma omp parallel for reduction(max : maxval)
    for (int i = 0; i < n; i++) {
        if (arr[i] > maxval) maxval = arr[i];
    }
    return maxval;
}
  --> same logic but reduction(max : maxval)


--- SUM ---

int sum(int arr[], int n) {
    int sum = 0;
    #pragma omp parallel for reduction(+ : sum)
    for (int i = 0; i < n; i++) {
        sum += arr[i];
    }
    return sum;
}
  --> reduction(+ : sum):
      Each thread gets private sum = 0
      Each adds its portion
      OpenMP adds all partial sums together


--- AVERAGE ---

double average(int arr[], int n) {
    return (double)sum(arr, n) / n;
}
  --> calls sum() (already parallel) and divides by n
  --> (double) = cast so we get decimal answer


--- MAIN ---

int main() {
    int n = 5;
    int arr[] = {1, 2, 3, 4, 5};

    cout << "The minimum value is: " << minval(arr, n) << endl;
      --> output: 1

    cout << "The maximum value is: " << maxval(arr, n) << endl;
      --> output: 5

    cout << "The summation is: " << sum(arr, n) << endl;
      --> output: 15

    cout << "The average is: " << average(arr, n) << endl;
      --> output: 3

    return 0;
}

AVAILABLE REDUCTION OPERATORS:
  +   (addition)
  -   (subtraction)
  *   (multiplication)
  min (minimum)
  max (maximum)
  &   (bitwise AND)
  |   (bitwise OR)
  ^   (bitwise XOR)

COMPILE: g++ -fopenmp minmax.cpp -o minmax



===============================================================
  PRACTICAL 4: PARALLEL LINEAR REGRESSION (AIML)
===============================================================

AIM: Compute best-fit line y = mx + c for 1 million data points
     using OpenMP to speed up the calculation.

WHAT IS LINEAR REGRESSION?
  Finding a straight line that best fits data points.
  m = slope (how steep the line is)
  c = intercept (where line crosses y-axis)

FORMULA:
  m = (n * sum_xy - sum_x * sum_y) / (n * sum_x2 - sum_x * sum_x)
  c = (sum_y - m * sum_x) / n

  We need 4 sums: sum_x, sum_y, sum_xy, sum_x2
  All 4 can be computed in parallel using reduction!


----------- CODE LINE BY LINE -----------

#include <iostream>
#include <vector>
#include <cstdlib>
#include <omp.h>
using namespace std;

int main() {
    int n = 1000000;
      --> 1 million data points

    vector<double> x(n), y(n);
      --> two arrays of 1 million elements

    #pragma omp parallel for
    for (int i = 0; i < n; i++) {
        x[i] = i * 0.001;
        y[i] = 2 * x[i] + 3;
    }
      --> generates test data: y = 2x + 3
      --> we KNOW the answer: slope=2, intercept=3
      --> even data generation is parallel (1M points!)
      --> x: 0, 0.001, 0.002, 0.003, ...
      --> y: 3, 3.002, 3.004, 3.006, ...

    double sum_x = 0, sum_y = 0, sum_xy = 0, sum_x2 = 0;
      --> initialize 4 sums

    #pragma omp parallel for reduction(+:sum_x, sum_y, sum_xy, sum_x2)
    for (int i = 0; i < n; i++) {
        sum_x  += x[i];
        sum_y  += y[i];
        sum_xy += x[i] * y[i];
        sum_x2 += x[i] * x[i];
    }
      --> THIS IS THE KEY LINE!
      --> reduction(+:sum_x, sum_y, sum_xy, sum_x2):
          Each thread gets PRIVATE copies of all 4 variables
          Each thread adds its portion
          After loop, OpenMP adds all partial sums
      --> 4 reductions in ONE loop - very efficient!

    double m = (n * sum_xy - sum_x * sum_y) / (n * sum_x2 - sum_x * sum_x);
    double c = (sum_y - m * sum_x) / n;
      --> applies the linear regression formula
      --> this part is sequential (just 2 calculations)

    cout << "Slope (m): " << m << endl;
    cout << "Intercept (c): " << c << endl;
      --> should print: Slope (m) = 2, Intercept (c) = 3
      --> matches our equation y = 2x + 3!

    return 0;
}

COMPILE: g++ -fopenmp aiml.cpp -o aiml

            QUICK COMPARISON OF ALL 4 HPC PRACTICALS

  Practical 1 - BFS/DFS:
    Key pragma: parallel for + critical + task + single
    Purpose: safe parallel graph traversal
